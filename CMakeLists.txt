cmake_minimum_required(VERSION 3.24)
project(roadwars LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Match the static release CRT used by official godot-cpp Windows builds.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
option(ROADWARS_BUILD_GODOT "Build the Godot GDExtension" OFF)
include(CTest)
add_subdirectory(game_src)
if(ROADWARS_BUILD_GODOT)
  add_subdirectory(platform/godot)
endif()
if(BUILD_TESTING)
  add_executable(roadwars_core_test tests/core/runtime_test.cpp)
  target_link_libraries(roadwars_core_test PRIVATE roadwars_core)
  add_test(NAME core COMMAND roadwars_core_test)
  add_executable(roadwars_entry_test tests/core/game_entry_test.cpp)
  target_link_libraries(roadwars_entry_test PRIVATE roadwars_core)
  add_test(NAME entry_loop COMMAND roadwars_entry_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(roadwars_not_implemented_test tests/core/not_implemented_test.cpp)
  target_link_libraries(roadwars_not_implemented_test PRIVATE roadwars_core)
  target_compile_definitions(roadwars_not_implemented_test PRIVATE NDEBUG)
  add_test(NAME unimplemented_trace COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/test_not_implemented.py $<TARGET_FILE:roadwars_not_implemented_test>)
  add_test(NAME scripts COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests/scripts WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
  add_test(NAME boundaries COMMAND ${Python3_EXECUTABLE} scripts/check_boundaries.py WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
  add_test(NAME progress COMMAND ${Python3_EXECUTABLE} scripts/progress.py check WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
  add_test(NAME generated_classes COMMAND ${Python3_EXECUTABLE} scripts/generate_classes.py --check WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
endif()
